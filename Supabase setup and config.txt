You will need to generate a supabase table and buctket 
For the bucket, just create one and name it as "generated-documents"
There is a python script under scripts/setup_supabase.py. But i think its a bit conviluted and might as well just create one manually. 

For the tables, run the following script:
DO NOT SELECT THE RSL OPTION> GENERATE WITHOUT RSL


-- runs table
CREATE TABLE runs (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  status text NOT NULL DEFAULT 'running',
  total_links integer,
  source_csv text,
  created_at timestamptz NOT NULL DEFAULT now(),
  finished_at timestamptz
);

-- jobs table
CREATE TABLE jobs (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  run_id uuid REFERENCES runs(id) ON DELETE CASCADE,
  link text,
  status text NOT NULL DEFAULT 'pending'
    CHECK (status IN ('pending','extracted','resume_generated','cover_letter_generated','completed','failed')),
  job_scope jsonb,
  resume_json jsonb,
  cover_letter_json jsonb,
  resume_url text,
  cover_letter_url text,
  error_message text,
  updated_at timestamptz,
  created_at timestamptz NOT NULL DEFAULT now()
);
